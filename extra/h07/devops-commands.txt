# Практична робота — DevOps: процеси
# Виконуйте в терміналі Linux / Ubuntu (не в PowerShell Windows).

# 1. Програма на диску та два запущені процеси
which sleep
ls -l $(which sleep)
sleep 60 &
sleep 60 &
ps aux | grep "sleep 60" | grep -v grep

# 2. Прочитати вивід ps
ps aux | head -1
ps aux | head -10
ps -p $$

# 3. Зупинити процес
jobs
kill %1
jobs
sleep 300 &
ps aux | grep "sleep 300" | grep -v grep
# Візьміть PID із виводу попередньої команди та підставте замість <PID>:
kill -9 <PID>
jobs

# 4. Передній план і фон
sleep 30
# Натисніть Ctrl+C після завершення спостереження
sleep 30 &
jobs
fg
# Натисніть Ctrl+Z
jobs
bg
jobs

# 5. Запустити HTTP-сервер із каталогу, в якому є файли
cd ~/dev/files-lab
python3 -m http.server 8000
# В іншому терміналі: ps aux | grep http.server | grep -v grep
# Поверніться до першого термінала та натисніть Ctrl+C
python3 -m http.server 8000 &
jobs
kill %1

# 6. Пошук за назвою процесу
sleep 500 &
sleep 500 &
pgrep -f "sleep 500"
pkill -f "sleep 500"
pgrep -f "sleep 500"

# 7. Навантаження системи
top
# Вийти з top клавішею q
