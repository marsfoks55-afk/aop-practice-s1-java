# Практична робота — Git: читання історії
# Команди запускайте в Git Bash у корені свого репозиторію.

# 1. Три вигляди історії
git log
git log --oneline -5
git log --oneline --graph --decorate -20

# 2. Фільтрування історії
git log -5 --stat
git log --author="Безкоровайний" --oneline
git log --since="2 weeks ago" --oneline
git log --oneline -- src/main/java/homework/h07/T1.java

# 3. Показати коміт
 git log --oneline -5
# Підставте реальний короткий хеш із виводу:
git show <hash>
git show HEAD~2

# 4. Автори рядків
git blame readme.md | head -20
# Візьміть хеш із виводу git blame і виконайте:
git show <hash>

# 5. Знайти коміти, у яких змінилася кількість входжень фрагмента
git log -S "homework.h07" --oneline

# 6. Створити тег (лише якщо тег v1.0.0 ще не існує)
git tag v1.0.0
git tag -l
git show v1.0.0
git log --oneline --decorate -3

# 7. Форматований лог
git log --format='%h | %ad | %an | %s' --date=short -10
git log --format='%C(yellow)%h%C(reset) %s %C(dim)(%ar)%C(reset)' -10

# 8. Історія за авторами й граф усіх гілок
git shortlog -sn
git log --oneline --all --graph -30
